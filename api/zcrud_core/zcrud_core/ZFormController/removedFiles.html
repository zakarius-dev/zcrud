Map<String, List<Object>> get removedFiles =>
    Map<String, List<Object>>.unmodifiable(<String, List<Object>>{
      for (final e in _removedFiles.entries)
        if (e.value.isNotEmpty) e.key: List<Object>.unmodifiable(e.value),
    });
