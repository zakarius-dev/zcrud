Future<void> stopNarration() async {
  final ZChatSpeechPort? port = speech;
  if (port != null) {
    try {
      await port.stop();
    } catch (_) {
      // Un moteur d'hôte qui lève à l'arrêt ne doit pas laisser l'interface
      // bloquée sur « lecture en cours ».
    }
  }
  _speaking.value = false;
}
