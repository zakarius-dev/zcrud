static String mapDocumentStatus(Object? legacy) {
  if (legacy is! String) return 'uploading';
  switch (legacy) {
    case 'uploading':
      return 'uploading';
    case 'converting':
    case 'embedding':
      return 'validating';
    case 'uploaded':
    case 'converted':
    case 'embedded':
      return 'ready';
    default:
      return 'uploading';
  }
}
