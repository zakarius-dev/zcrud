Propagation PAR LOT BORNÉE (AD-9) d'un changeset d'entries, chacune écrite verbatim (méta préservée, jamais now()). Le changeset est découpé en lots de ≤ kMaxBatchWrites (450), chaque lot étant un WriteBatch committé atomiquement (aucune écriture partielle non-commit). Liste vide → Right(unit). FirebaseException → Left(ZServerFailure).

Future<ZResult<Unit>> applyMergedAll(List<ZSyncEntry<T>> entries) =>
    _guard(() async {
      for (var start = 0;
          start < entries.length;
          start += kMaxBatchWrites) {
        final end = (start + kMaxBatchWrites < entries.length)
            ? start + kMaxBatchWrites
            : entries.length;
        final batch = _firestore.batch();
        for (var i = start; i < end; i++) {
          final entry = entries[i];
          final id = entry.entity.id;
          if (id == null) {
            return Left<ZFailure, Unit>(
              ZDomainFailure(
                'applyMergedAll: entité éphémère (id null) interdite '
                '(kind=$_kind)',
              ),
            );
          }
          batch.set(_rawCollection().doc(id), _mergedMap(entry, id));
        }
        await batch.commit();
      }
      return Right<ZFailure, Unit>(unit);
    });
